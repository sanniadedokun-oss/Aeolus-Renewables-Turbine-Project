def analyse_feature(column, label, unit='', investigation_rule=None):
    """Display a histogram, statistical summary, and an EDA interpretation for one feature."""
    values = df[column].dropna()
    quantiles = values.quantile([0.01, 0.05, 0.25, 0.50, 0.75, 0.95, 0.99])
    summary = pd.DataFrame({
        'count': [values.count()], 'missing': [df[column].isna().sum()], 'minimum': [values.min()],
        'p01': [quantiles.loc[0.01]], 'p05': [quantiles.loc[0.05]], 'p25': [quantiles.loc[0.25]],
        'median': [quantiles.loc[0.50]], 'mean': [values.mean()], 'p75': [quantiles.loc[0.75]],
        'p95': [quantiles.loc[0.95]], 'p99': [quantiles.loc[0.99]], 'maximum': [values.max()],
        'standard_deviation': [values.std()], 'skewness': [values.skew()]
    }, index=[column]).round(3)
    counts, edges = np.histogram(values, bins=50)
    peak_index = counts.argmax()
    peak_interval = f'{edges[peak_index]:.3g} to {edges[peak_index + 1]:.3g}'
    skewness = values.skew()
    shape = 'approximately symmetric' if abs(skewness) < 0.3 else ('right-skewed' if skewness > 0 else 'left-skewed')
    central_range = f'{quantiles.loc[0.05]:.3g} to {quantiles.loc[0.95]:.3g}'
    tail_count = ((values < quantiles.loc[0.01]) | (values > quantiles.loc[0.99])).sum()

    display(Markdown(f'## {label} (`{column}`)'))
    display(summary)
    fig, ax = plt.subplots(figsize=(9, 4.5))
    sns.histplot(values, bins=50, kde=True, color='#4C78A8', ax=ax)
    ax.axvline(quantiles.loc[0.50], color='#E45756', linestyle='--', label=f'Median: {quantiles.loc[0.50]:.3g}')
    ax.set(title=f'Distribution of {label}', xlabel=f'{label} ({unit})' if unit else label, ylabel='Observations')
    ax.legend()
    plt.tight_layout()
    plt.show()
    plt.close(fig)

    rule_message = investigation_rule(values, quantiles) if investigation_rule else None
    interpretation = [
        f'Most observations are concentrated around **{peak_interval} {unit}** (the most populated histogram bin); the middle 90% span **{central_range} {unit}**.',
        f'The distribution is **{shape}** (skewness = **{skewness:.2f}**), with an observed range of **{values.min():.3g} to {values.max():.3g} {unit}**.',
        f'**{tail_count:,}** records ({tail_count / len(values):.2%}) lie outside the 1st–99th percentile interval and merit contextual review alongside turbine operating state.'
    ]
    if rule_message:
        interpretation.append(rule_message)
    display(Markdown('**Key observations:**\n' + '\n'.join(f'- {item}' for item in interpretation)))

def wind_rule(values, q):
    return (f'No negative wind speeds are present; the maximum is **{values.max():.3g} m/s**. Values above 25 m/s are high but can occur during strong-wind conditions and should be checked against cut-out/control status.' if (values < 0).sum() == 0 else 'Negative wind speeds are present and should be investigated.')

def density_rule(values, q):
    outside = ((values < 1.0) | (values > 1.4)).sum()
    return f'**{outside:,}** values fall outside the broad 1.0–1.4 kg/m³ screening range; investigate them with weather and sensor-quality context.' if outside else 'All values fall within the broad 1.0–1.4 kg/m³ screening range; no immediately implausible readings are apparent.'

def upper_temperature_rule(threshold, name):
    return lambda values, q: f'**{(values > threshold).sum():,}** readings exceed **{threshold}°C**; these high {name} observations deserve review against load, ambient temperature, and failure status.'

def nonnegative_rule(name):
    return lambda values, q: f'No negative {name} values are present.' if (values < 0).sum() == 0 else f'Negative {name} values are present and should be investigated.'
